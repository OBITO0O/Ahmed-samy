<script>
  // محاكي بيانات من السيرفر
  const userData = {
    username: "يوسف جمال",
    score: 256,
    days: 42,
    profileImage: "https://via.placeholder.com/150/FFD700/0D1B2A?text=User"
  };

  // تحميل البيانات
  document.getElementById('username').textContent = userData.username;
  document.getElementById('score').textContent = userData.score;
  document.getElementById('days').textContent = userData.days;
  document.getElementById('profileImage').src = userData.profileImage;

  function shareScore(){
    const shareText = `لقد حصلت على ${userData.score} نقطة في تطبيق الرفيق!`;
    if(navigator.share){
      navigator.share({
        title: 'إنجازاتي',
        text: shareText,
        url: window.location.href
      }).then(()=> console.log('Shared successfully'))
      .catch(err => console.error(err));
    } else {
      alert("نسخ الرابط: " + window.location.href);
    }
  }
</script>
